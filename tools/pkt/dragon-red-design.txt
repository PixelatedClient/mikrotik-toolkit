// The dragon, after the reference: an upright red pixel dragon with a big horned head, cream chest, a small folded arm,
// a curled tail and small spiky wings, black outline. One unit = 8 px, it faces right, and it is drawn small because it sits far away.
const dpal: Record<string, string> = {
  k: '#1a0d0d', r: '#e2452b', R: '#b3301f', c: '#f6d98b', C: '#dba95a', b: '#7a5230', B: '#4d3320', w: '#ffffff', p: '#a01f30', o: '#ff9a3a', y: '#ffd23f', s: '#cfd6ff',
};
const bmp = (rows: string[], ox: number, oy: number) => {
  const out: { x: number; y: number; w: number; h: number; fill: string }[] = [];
  rows.forEach((row, y) => {
    let x = 0;
    while (x < row.length) {
      const ch = row[x];
      if (ch === '.' || !dpal[ch]) { x++; continue; }
      let e = x;
      while (e + 1 < row.length && row[e + 1] === ch) e++;
      out.push({ x: (ox + x) * P, y: (oy + y) * P, w: (e - x + 1) * P, h: P, fill: dpal[ch] });
      x = e + 1;
    }
  });
  return out;
};
